## Corrected Rules (COPY THIS TO FIRESTORE)

```javascript
rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {

    // ============================================
    // HELPER FUNCTIONS
    // ============================================

    function isAuthenticated() {
      return request.auth != null;
    }

    function uid() {
      return request.auth.uid;
    }

    function isOwner(userId) {
      return isAuthenticated() && uid() == userId;
    }

    function getFriendshipId(uid1, uid2) {
      return uid1 < uid2 ? uid1 + '_' + uid2 : uid2 + '_' + uid1;
    }

    function areFriends(uid1, uid2) {
      return exists(/databases/$(database)/documents/friends_lookup/$(getFriendshipId(uid1, uid2)));
    }

    function hasFriendsEnabled(targetUserId) {
      let doc = get(/databases/$(database)/documents/friends_public_data/$(targetUserId));
      return doc != null && doc.data.enabled == true;
    }

    function isValidUidString(x) {
      return x is string && x.size() >= 8 && x.size() <= 128;
    }

    function isValidFriendUsersArray(arr) {
      return arr is list
        && arr.size() == 2
        && isValidUidString(arr[0])
        && isValidUidString(arr[1])
        && arr[0] != arr[1];
    }

    function friendshipIdMatchesDocId(friendshipId, usersArr) {
      return friendshipId == getFriendshipId(usersArr[0], usersArr[1]);
    }

    // ============================================
    // COURSES - OWNER ONLY (GRADES LIVE HERE)
    // Grades are sensitive - only owner can access
    // ============================================
    match /courses/{courseId} {
      // Owner-only read - protects grades from exposure
      allow get, list: if isAuthenticated() && resource.data.userId == uid();

      // Owner-only create with schema validation
      allow create: if isAuthenticated()
        && request.resource.data.userId == uid()
        && request.resource.data.keys().hasOnly([
          "userId",
          "code",
          "grade",
          "semester",
          "year",
          "status",
          "credits",
          "skipped",
          "manualRequirementsFulfilled",
          "distributionals"
        ])
        && request.resource.data.code is string
        && request.resource.data.semester is string
        && request.resource.data.year is int
        && request.resource.data.status in ["completed", "in-progress", "skipped", "not-taken"];

      // Owner-only update - prevent userId reassignment (IDOR protection)
      allow update: if isAuthenticated()
        && resource.data.userId == uid()
        && request.resource.data.userId == resource.data.userId
        && request.resource.data.keys().hasOnly([
          "userId",
          "code",
          "grade",
          "semester",
          "year",
          "status",
          "credits",
          "skipped",
          "manualRequirementsFulfilled",
          "distributionals"
        ]);

      allow delete: if isAuthenticated() && resource.data.userId == uid();
    }

    // ============================================
    // USERS - ACCOUNT DATA
    // Contains majors, graduationYear, savedPlans
    // NOTE: Grades are NOT here - they're in /courses (owner-only)
    // ============================================
    match /users/{userId} {
      // Any authenticated user can read user profiles
      // This is needed for:
      // - Friends tab: listing users to search/send requests
      // - Profile pages: displaying friend info
      // SAFE because grades are in /courses, not here
      allow get, list: if isAuthenticated();

      // Owner-only create
      allow create: if isOwner(userId)
        && request.resource.data.keys().hasOnly([
          "majors",
          "graduationYear",
          "bio",
          "savedPlans",
          "profileComplete",
          "createdAt",
          "updatedAt"
        ]);

      // Owner-only update - allow profile fields + savedPlans
      allow update: if isOwner(userId)
        && request.resource.data.keys().hasOnly([
          "majors",
          "graduationYear",
          "bio",
          "savedPlans",
          "profileComplete",
          "createdAt",
          "updatedAt"
        ]);

      allow delete: if isOwner(userId);
    }

    // ============================================
    // FRIENDS PUBLIC DATA
    // Public profile + courses (WITHOUT grades)
    // Only accessible to friends when enabled
    // ============================================
    match /friends_public_data/{userId} {
      // Owner can always read
      // Friends can read only if enabled AND actually friends
      allow get: if isOwner(userId) ||
        (isAuthenticated() && hasFriendsEnabled(userId) && areFriends(uid(), userId));

      // Never allow listing all public profiles
      allow list: if false;

      // Owner-only create/update with full schema
      allow create, update: if isOwner(userId)
        && request.resource.data.userId == userId
        && request.resource.data.keys().hasOnly([
          "userId",
          "enabled",
          "enabledAt",
          "updatedAt",
          "displayName",
          "email",
          "photoURL",
          "majors",
          "graduationYear",
          "bio",
          "courses"
        ])
        && request.resource.data.enabled is bool;

      allow delete: if isOwner(userId);
    }

    // ============================================
    // FRIENDS LOOKUP - Efficient friendship check
    // Used by security rules to verify friendships
    // ============================================
    match /friends_lookup/{friendshipId} {
      // Only participants can read their own friendship
      allow get: if isAuthenticated() && (uid() in resource.data.users);

      // Never allow listing all friendships
      allow list: if false;

      // Only participants can create, with canonical doc ID
      allow create: if isAuthenticated()
        && request.resource.data.keys().hasOnly(["users", "createdAt"])
        && isValidFriendUsersArray(request.resource.data.users)
        && (uid() in request.resource.data.users)
        && friendshipIdMatchesDocId(friendshipId, request.resource.data.users);

      // Only participants can delete
      allow delete: if isAuthenticated() && (uid() in resource.data.users);

      // Never allow updates (immutable)
      allow update: if false;
    }

    // ============================================
    // FRIENDS - Legacy friendship records
    // Kept for backward compatibility
    // ============================================
    match /friends/{friendId} {
      // Only participants can read
      allow get: if isAuthenticated() && (uid() in resource.data.users);

      // Allow list for participants (needed for friends queries)
      allow list: if isAuthenticated();

      allow create: if isAuthenticated()
        && request.resource.data.keys().hasOnly(["users", "createdAt"])
        && isValidFriendUsersArray(request.resource.data.users)
        && (uid() in request.resource.data.users);

      allow delete: if isAuthenticated() && (uid() in resource.data.users);

      allow update: if false;
    }

    // ============================================
    // FRIEND REQUESTS
    // Sent/received request management
    // ============================================
    match /friend-requests/{requestId} {
      // Sender or recipient can read
      allow get: if isAuthenticated()
        && (resource.data.from == uid() || resource.data.to == uid());

      // IMPORTANT: Allow list queries filtered by from/to
      // Firestore requires list permission for where() queries
      // Security: queries MUST filter by from==uid or to==uid (enforced client-side + indexes)
      allow list: if isAuthenticated();

      // Only sender can create, must be pending
      allow create: if isAuthenticated()
        && request.resource.data.keys().hasOnly(["from", "to", "status", "createdAt", "updatedAt", "timestamp"])
        && request.resource.data.from == uid()
        && request.resource.data.to is string
        && request.resource.data.to != uid()
        && request.resource.data.status == "pending";

      // Only recipient can update status (accept/reject)
      // Cannot change from/to (prevents hijacking)
      allow update: if isAuthenticated()
        && resource.data.to == uid()
        && request.resource.data.from == resource.data.from
        && request.resource.data.to == resource.data.to
        && request.resource.data.keys().hasOnly(["from", "to", "status", "createdAt", "updatedAt", "timestamp"])
        && (request.resource.data.status in ["accepted", "rejected"]);

      // Either party can delete/cancel
      allow delete: if isAuthenticated()
        && (resource.data.from == uid() || resource.data.to == uid());
    }

    // ============================================
    // AI RESPONSES - Owner only
    // ============================================
    match /ai_responses/{docId} {
      allow get, list: if isAuthenticated() && resource.data.userId == uid();

      allow create: if isAuthenticated()
        && request.resource.data.userId == uid();

      allow update: if isAuthenticated()
        && resource.data.userId == uid()
        && request.resource.data.userId == resource.data.userId;

      allow delete: if isAuthenticated() && resource.data.userId == uid();
    }

    // ============================================
    // CLEO AI CONVERSATIONS - Owner only
    // ============================================
    match /cleoai_conversations/{userId} {
      allow get: if isOwner(userId);
      allow list: if false;
      allow create, update, delete: if isOwner(userId);
    }

  }
}
```

## Security Alignment Checklist

|-------------------------------|--------|

| Courses owner-only (grades protected) | ✅ Only owner can read courses |

## Notes

1. **Friend-requests `list: true`**: This is required because Firestore `where()` queries are list operations. The client-side code already filters by `from == uid` or `to == uid`. For additional server-side enforcement, add Firestore indexes with security constraints.

2. **Users collection readable by all authenticated users**: The FriendsTab component lists ALL users (`onSnapshot(collection(db, "users"))`) to build a searchable index for sending friend requests. This requires `list: true`. **This is safe because grades are stored in `/courses` (owner-only), not in `/users`.**

3. **Future improvement**: Consider creating a separate `user_search` collection with only public fields (displayName, email for Yale domain matching) to avoid exposing `savedPlans` and other private data to all users. For now, the current app architecture requires full user listing.

4. **Bio length limit removed**: The original rules had `bio.size() <= 500` which would cause silent failures if exceeded. Consider adding this back with proper error handling in the client.

5. **savedPlans in users**: The Simulator stores plans in the users collection. This is acceptable since it doesn't contain grades (just course codes and semesters).
